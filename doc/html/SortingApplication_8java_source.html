236 private void countOccurrencesParallel(Car[] cars, int targetPower) {
237 int processors = Runtime.getRuntime().availableProcessors(); // Получаем количество процессоров
238 ExecutorService executorService = Executors.newFixedThreadPool(processors); // Создаем пул потоков
239
240 int chunkSize = Math.max(1, cars.length / processors); // Количество элементов в каждом потоке
241 Counter counter = new Counter(); // Счетчик для подсчета количества элементов с заданной мощностью
242
243 for (int i = 0; i < cars.length; i += chunkSize) { // Разбиваем массив на части
244 int start = i;
245 int end = Math.min(i + chunkSize, cars.length);
246
247 executorService.submit(() -> { // Запускаем поток
248 int localCount = 0;
249 for (int j = start; j < end; j++) { // Проходим по части массива и считаем количество элементов с заданной мощностью
250 if (cars[j].getPower() == targetPower) { // Проверяем мощность элемента
251 localCount++;
252 }
253 }
254 synchronized (counter) { // Синхронизируем доступ к общему ресурсу
255 counter.value += localCount;
256 }
257 });
258 }
259 executorService.shutdown(); // Останавливаем потоки
260 try {
261 executorService.awaitTermination(Long.MAX_VALUE, TimeUnit.NANOSECONDS); // Ожидаем завершения всех потоков
262 } catch (InterruptedException e) {
263 Thread.currentThread().interrupt(); // Восстанавливаем прерванность потока
264 }
265 System.out.println("Количество элементов с мощностью " + targetPower + ": " + counter.value);
266 }
